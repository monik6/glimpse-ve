function insertNewSlides(layoutIndexes, numSlides) {
  const presentation = SlidesApp.getActivePresentation();
  const master = presentation.getMasters()[0]; // assuming the layout is in the first master
  const layouts = master.getLayouts();
  const slides = [];
  
  for (let i = 0; i < numSlides; i++) {
    const layoutIndex = layoutIndexes[i];
    const slide = presentation.appendSlide(layouts[layoutIndex]);

   slides.push(slide);
  }
  
  buttonSlideRelation(numSlides);
  
  return slides;
}

function insertNextSlideButton(startIndex, endIndex, shapeText, left) {
  var presentation = SlidesApp.getActivePresentation();
  
  for (var i = startIndex; i <= endIndex; i++) {
    var slide = presentation.getSlides()[i];
    var shape = slide.insertShape(SlidesApp.ShapeType.ROUND_RECTANGLE, left, 344, 115, 34);
    

    var textRange = shape.getText();
    textRange.setText(shapeText);
    shape.getText().getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER);
    var textFormat = textRange.getTextStyle();
    textFormat.setBold(true);
    textFormat.setFontFamily('Montserrat');
    
    shape.setLinkSlide(presentation.getSlides()[i+1]);
  }
}

function insertSpecificSlideButton(startIndex, endIndex, shapeText, slideLinked, left) {
  var presentation = SlidesApp.getActivePresentation();
  
  for (var i = startIndex; i <= endIndex; i++) {
    var slide = presentation.getSlides()[i];
    var shape = slide.insertShape(SlidesApp.ShapeType.ROUND_RECTANGLE, left, 344, 115, 34);

    var textRange = shape.getText();
    textRange.setText(shapeText);
    shape.getText().getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER);
    var textFormat = textRange.getTextStyle();
    textFormat.setBold(true);
    textFormat.setFontFamily('Montserrat');
    
    shape.setLinkSlide(presentation.getSlides()[slideLinked]);
  }
}

function insertRoomSlideButton(corridorSlide, roomLinked, top){
  var presentation = SlidesApp.getActivePresentation();
    var slide = presentation.getSlides()[corridorSlide];
    var shape = slide.insertShape(SlidesApp.ShapeType.ROUND_RECTANGLE, 67, top, 83, 22);
    shape.getFill().setTransparent();
    shape.getBorder().setTransparent();
    shape.setLinkSlide(presentation.getSlides()[roomLinked]);
}
