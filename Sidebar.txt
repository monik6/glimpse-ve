// JavaScript code for two slideshows
let slideIndex1 = 0;
let slideIndex2 = 0;
showSlides1();
showSlides2();

function showSlides1() {
  let i;
  const slides = document.getElementsByClassName("slide1");
  for (i = 0; i < slides.length; i++) {
    slides[i].style.display = "none";
  }
  slideIndex1++;
  if (slideIndex1 > slides.length) {
    slideIndex1 = 1;
  }
  slides[slideIndex1 - 1].style.display = "block";
  setTimeout(showSlides1, 3000); // Change image every 3 seconds
}

function showSlides2() {

<script>
// Select all task headers
const taskHeaders = document.querySelectorAll('.task-header');

// Add click event listeners to each task header
taskHeaders.forEach((taskHeader) => {
  taskHeader.addEventListener('click', () => {
    // Toggle the 'active' class to change arrow icon
    taskHeader.classList.toggle('active');
    // Get the next element (task content) after the clicked header
    const taskContent = taskHeader.nextElementSibling;
    // If the task is currently open, close it by setting the max-height to 0
    if (taskContent.style.maxHeight) {
      taskContent.style.maxHeight = null;
    } else {
      // Otherwise, open the task by setting the max-height to the scroll height
      taskContent.style.maxHeight = taskContent.scrollHeight + '5px';
    }
  });
});

// Open Web App Button

function embedExhibition() {
  const link = document.getElementById("hiddenLink").textContent;
  const iframeCode = '<iframe src="' + link + '" frameborder="0" width="100%" height="1200"></iframe>';
  document.getElementById("embedCode").textContent = iframeCode;
  
  // Copy the iframe code to the clipboard
  navigator.clipboard.writeText(iframeCode);
  
  // Show an alert message using SlidesApp.getUi().alert()
  alert('Embed code copied to clipboard!');
}
