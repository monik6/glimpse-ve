function editExistingSlideFromSheet() {
  var sheetUrl1 = getSheetUrl1();
  const sheetId = sheetUrl1.split("/")[5];
  const sheet = SpreadsheetApp.openById(sheetId).getSheets()[0];
  const data = sheet.getDataRange().getValues();

  const presentation = SlidesApp.getActivePresentation();

  for (let i = 2; i < data.length; i++) {
    const row = data[i];
    const slideIndex = row[0] - 1;
    const slide = presentation.getSlides()[slideIndex];

    const projectTitle = row[1];
    if (projectTitle) {
      slide.getPlaceholder(SlidesApp.PlaceholderType.TITLE).asShape().getText().setText(projectTitle);
    }

    const projectDescription = row[2];
    if (projectDescription) {
      slide.getPlaceholder(SlidesApp.PlaceholderType.BODY).asShape().getText().setText(projectDescription);
    }

    const videoUrl = row[3];
    if (videoUrl) {
      var layout = slide.getLayout().getLayoutName();
      var tablePosition = {};
      if (layout == 'CUSTOM_3_1') {
        slide.insertVideo(videoUrl, 125, 125, 175, 125); 
        tablePosition = {left: 314, top: 210, width: 252, height: 30};
      } else if (layout == 'CUSTOM_3_1_1') {
        slide.insertVideo(videoUrl, 410, 155, 175, 125); 
        tablePosition = {left: 178, top: 210, width: 230, height: 30};
      } else if (layout == 'CUSTOM_3_1_1_1') {
        slide.insertVideo(videoUrl, 110, 125, 175, 125); 
        tablePosition = {left: 295, top: 210, width: 239, height: 30};
      } else if (layout == 'CUSTOM_3_1_1_1_1') {
        slide.insertVideo(videoUrl, 440, 125, 175, 125); 
        tablePosition = {left: 149, top: 210, width: 239, height: 30};
      } else if (layout == 'CUSTOM_3_1_1_1_1_1') {
        slide.insertVideo(videoUrl, 400, 155, 175, 125);
        tablePosition = {left: 184, top: 210, width: 205, height: 30};
      }

      var table = slide.insertTable(2, 2, tablePosition.left, tablePosition.top, tablePosition.width, tablePosition.height);
    }

    const teamMembers = row.slice(4, 8).filter(member => member !== '');
    for (let j = 0; j < 4; j++) {
      if (teamMembers[j]) {
        if (videoUrl) {
          table.getCell(Math.floor(j/2), j%2).getText().setText(teamMembers[j]);
        }
      }
    }
  }

  SlidesApp.getUi().alert('Data inserted successfully!');
}
